Estado, formulários e navegação

Aula · 14 min

StatefulWidget

class Contador extends StatefulWidget {
  const Contador({super.key});
  @override
  State<Contador> createState() => _ContadorState();
}

class _ContadorState extends State<Contador> {
  int hospedes = 1;

  @override
  Widget build(BuildContext context) {
    return Row(
      mainAxisSize: MainAxisSize.min,
      children: [
        IconButton(
          onPressed: hospedes > 1 ? () => setState(() => hospedes--) : null,   // null desabilita
          icon: const Icon(Icons.remove),
        ),
        Text('$hospedes hóspede(s)'),
        IconButton(onPressed: () => setState(() => hospedes++), icon: const Icon(Icons.add)),
      ],
    );
  }
}

setState avisa o Flutter que o estado mudou e o build roda de novo. Mudar a variável sem setState não atualiza a tela.

Formulários

final _form = GlobalKey<FormState>();
final _nome = TextEditingController();

Form(
  key: _form,
  child: Column(children: [
    TextFormField(
      controller: _nome,
      decoration: const InputDecoration(labelText: 'Nome'),
      validator: (v) => (v == null || v.trim().isEmpty) ? 'Informe o nome' : null,
    ),
    FilledButton(
      onPressed: () {
        if (_form.currentState!.validate()) print(_nome.text);
      },
      child: const Text('Reservar'),
    ),
  ]),
)

Controllers precisam ser liberados: sobrescreva dispose() no State e chame _nome.dispose().
