Desafio: ler argumentos de um script

Desafio · 18 min

Scripts de linha de comando recebem opções como node relatorio.js vendas.csv --mes 10 --formato=pdf --verbose. Escreva lerArgumentos(lista), que recebe a lista de argumentos (já sem os dois primeiros do process.argv) e devolve um objeto:

  • --nome valor e --nome=valor viram { nome: 'valor' }.
  • Uma opção sem valor (seguida de outra opção ou no fim) vira true: --verbose → { verbose: true }.
  • Valores que são números viram número: --mes 10 → { mes: 10 }.
  • O que não é opção vai, em ordem, para a lista _: { _: ['vendas.csv'] }. A lista sempre existe, mesmo vazia.

O que o código precisa ter

  • Argumentos soltos vão para _, em ordem
  • --nome valor e --nome=valor viram propriedades
  • Opção sem valor vira true
  • Números viram número
Ver uma solução possível 
<script>
  function converter(valor) {
    return valor !== '' && !Number.isNaN(Number(valor)) ? Number(valor) : valor;
  }

  function lerArgumentos(lista) {
    const resultado = { _: [] };
    for (let i = 0; i < lista.length; i++) {
      const arg = lista[i];
      if (!arg.startsWith('--')) {
        resultado._.push(arg);
        continue;
      }
      const [nome, valorJunto] = arg.slice(2).split(/=(.*)/);
      if (valorJunto !== undefined) {
        resultado[nome] = converter(valorJunto);
      } else if (i + 1 < lista.length && !lista[i + 1].startsWith('--')) {
        resultado[nome] = converter(lista[++i]);
      } else {
        resultado[nome] = true;
      }
    }
    return resultado;
  }

  console.log(lerArgumentos(['vendas.csv', '--mes', '10', '--formato=pdf', '--verbose']));
</script>
