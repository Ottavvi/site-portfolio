Aula · 13 min

Middlewares

Um middleware é uma função que roda antes (e às vezes depois) das rotas: registra o pedido, confere o login, lê o corpo, trata erros. No Express, cada um recebe (req, res, next) e chama next() para passar adiante, ou responde e encerra a cadeia: 

app.use((req, res, next) => {
  const inicio = Date.now();
  res.on('finish', () => console.log(req.method, req.url, res.statusCode, Date.now() - inicio + 'ms'));
  next();
});

function exigirLogin(req, res, next) {
  if (!req.headers.authorization) return res.status(401).json({ erro: 'Faça login' });
  next();
}

app.get('/api/pedidos', exigirLogin, (req, res) => { /* ... */ });

// Tratador de erros: 4 parâmetros, registrado por último
app.use((erro, req, res, next) => {
  console.error(erro);
  res.status(500).json({ erro: 'Erro interno' });
});

Testes com node:test

O Node já vem com um executor de testes. Arquivos *.test.js são encontrados sozinhos por node --test: 

// precos.test.js
import { test } from 'node:test';
import assert from 'node:assert/strict';
import { aplicarDesconto } from './precos.js';

test('10% de desconto', () => {
  assert.equal(aplicarDesconto(100, 0.1), 90);
});

test('desconto inválido lança erro', () => {
  assert.throws(() => aplicarDesconto(100, 2), /inválido/);
});

Com "test": "node --test" no package.json, npm test roda tudo, e um workflow do GitHub Actions pode rodar a cada push (veja a trilha de GitHub Actions).

Publicar

  • Plataformas como Render, Railway e Fly.io leem o seu repositório, rodam npm ci e npm start, e dão um endereço com HTTPS. Configure as variáveis de ambiente no painel delas.
  • Use a porta que a plataforma informa: process.env.PORT.
  • Num servidor próprio (VPS), um gerenciador como o PM2 ou o systemd reinicia o app se ele cair, e um proxy (Nginx, Caddy) cuida do HTTPS.
  • Front-end estático (como este site) não precisa de Node no servidor: GitHub Pages, Netlify e Vercel bastam. 
<script>
  // A ideia da cadeia de middlewares, em miniatura: cada um decide se chama o próximo
  const cadeia = [
    (ctx, proximo) => { ctx.log.push('registrar'); return proximo(); },
    (ctx, proximo) => (ctx.logado ? proximo() : 'parou: 401'),
    () => 'rota respondeu 200'
  ];
  function rodar(ctx, i = 0) {
    return cadeia[i](ctx, () => rodar(ctx, i + 1));
  }
  console.log(rodar({ log: [], logado: true }));
  console.log(rodar({ log: [], logado: false }));
</script>
