Desafio: cadeia de middlewares

Desafio · 25 min

Escreva compor(middlewares), que recebe uma lista de funções async (ctx, next) e devolve uma função async (ctx) que roda a cadeia:

  • Cada middleware roda na ordem e só chama o próximo se fizer await next(). Código depois do await next() roda na volta (como o Koa).
  • Um middleware que não chama next encerra a cadeia.
  • Chamar next() duas vezes no mesmo middleware lança um erro.

Depois, escreva o middleware exigirToken: se ctx.headers.authorization não for 'Bearer segredo', coloca ctx.status = 401 e ctx.corpo = { erro: 'Não autorizado' } e para a cadeia.

O que o código precisa ter

  • Os middlewares rodam na ordem, ida e volta
  • Quem não chama next encerra a cadeia
  • Chamar next duas vezes lança erro
  • exigirToken barra sem o token certo e deixa passar com ele
Ver uma solução possível 
<script>
  function compor(middlewares) {
    return async function (ctx) {
      async function rodar(i) {
        const middleware = middlewares[i];
        if (!middleware) return;
        let chamou = false;
        await middleware(ctx, async () => {
          if (chamou) throw new Error('next() chamado mais de uma vez');
          chamou = true;
          await rodar(i + 1);
        });
      }
      await rodar(0);
    };
  }

  async function exigirToken(ctx, next) {
    if (ctx.headers?.authorization !== 'Bearer segredo') {
      ctx.status = 401;
      ctx.corpo = { erro: 'Não autorizado' };
      return;
    }
    await next();
  }

  const app = compor([
    async (ctx, next) => {
      const inicio = performance.now();
      await next();
      console.log(ctx.status, Math.round(performance.now() - inicio) + 'ms');
    },
    exigirToken,
    async ctx => {
      ctx.status = 200;
      ctx.corpo = { pedidos: [] };
    }
  ]);

  app({ headers: { authorization: 'Bearer segredo' } });
</script>
