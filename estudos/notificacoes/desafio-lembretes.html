Desafio: lembretes agendados

Desafio · 22 min

Monte o agendador de lembretes de um app de tarefas. A função avisar(texto) já existe (ela mostra um toast).

  • agendarLembrete(texto, segundos): agenda o aviso, mostra o lembrete em #lembretes (um <li> com o texto) e devolve um id.
  • Quando chega a hora: chama avisar(texto) e tira o <li> da lista.
  • cancelarLembrete(id): o aviso não acontece mais e o <li> sai da lista.

O que o código precisa ter

  • agendarLembrete mostra o lembrete na lista e devolve um id
  • Na hora, chama avisar e tira o lembrete da lista
  • cancelarLembrete impede o aviso e tira da lista
Ver uma solução possível 
<ul id="lembretes"></ul>
<div id="avisos" role="status"></div>

<script>
  function avisar(texto) {
    const div = document.createElement('div');
    div.className = 'toast';
    div.textContent = texto;
    document.querySelector('#avisos').append(div);
  }

  const agendados = new Map(); // id do timer → <li> na lista

  function agendarLembrete(texto, segundos) {
    const li = document.createElement('li');
    li.textContent = texto;
    document.querySelector('#lembretes').append(li);

    const id = setTimeout(() => {
      avisar(texto);
      li.remove();
      agendados.delete(id);
    }, segundos * 1000);

    agendados.set(id, li);
    return id;
  }

  function cancelarLembrete(id) {
    clearTimeout(id);
    agendados.get(id)?.remove();
    agendados.delete(id);
  }

  agendarLembrete('Tomar água', 2);
</script>
