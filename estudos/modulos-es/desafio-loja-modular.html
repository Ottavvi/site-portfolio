Desafio: carrinho em módulos

Desafio · 20 min

O carrinho funciona, mas está tudo em main.js. Reorganize em módulos:

  • dinheiro.js exporta (com nome) a função formatar(valor), que devolve o valor em reais com Intl.NumberFormat.
  • carrinho.js tem como exportação padrão a função que calcula o total dos itens (preço vezes quantidade).
  • main.js importa as duas e mostra o total formatado em #total, sem definir nenhuma das funções.

O que o código precisa ter

  • dinheiro.js exporta a função formatar
  • carrinho.js tem uma exportação padrão
  • main.js importa dos dois módulos e não define as funções
  • #total mostra R$ 160,30
Ver uma solução possível 
<p>Total do carrinho: <strong id="total"></strong></p>
<script type="module" src="./main.js"></script>

// --- dinheiro.js ---
const moeda = new Intl.NumberFormat('pt-BR', { style: 'currency', currency: 'BRL' });

export function formatar(valor) {
  return moeda.format(valor);
}

// --- carrinho.js ---
export default function total(lista) {
  return lista.reduce((soma, item) => soma + item.preco * item.quantidade, 0);
}

// --- main.js ---
import { formatar } from './dinheiro.js';
import total from './carrinho.js';

const itens = [
  { nome: 'Café', preco: 62.9, quantidade: 2 },
  { nome: 'Caneca', preco: 34.5, quantidade: 1 }
];

document.querySelector('#total').textContent = formatar(total(itens));
