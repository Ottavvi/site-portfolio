Desafio: sistema de tradução

Desafio · 25 min

Escreva o pequeno sistema de tradução do site:

  • t(idioma, chave, vars): devolve o texto da chave no idioma, trocando cada {nome} por vars.nome.
  • Se o valor da chave for um objeto de plurais (como carrinho), escolha a forma com Intl.PluralRules usando vars.n (e use other se a categoria não existir).
  • Se a chave não existir no idioma, use o português; se não existir em lugar nenhum, devolva a própria chave.
  • aplicar(idioma): preenche cada elemento com data-i18n e muda o lang do <html>.

O que o código precisa ter

  • Troca variáveis: t("pt", "saudacao", { nome: "Ana" })
  • Escolhe a forma do plural pelo número
  • Cai para o português e depois para a chave
  • aplicar preenche os [data-i18n] e muda o lang
Ver uma solução possível 
<h1 data-i18n="titulo"></h1>
<p data-i18n="rodape"></p>

<script>
  const textos = {
    pt: {
      titulo: 'Minha loja',
      saudacao: 'Olá, {nome}!',
      carrinho: { one: '{n} item no carrinho', other: '{n} itens no carrinho' },
      rodape: 'Feito com café'
    },
    en: {
      titulo: 'My shop',
      saudacao: 'Hi, {nome}!',
      carrinho: { one: '{n} item in your cart', other: '{n} items in your cart' }
    }
  };

  function t(idioma, chave, vars = {}) {
    let valor = textos[idioma]?.[chave] ?? textos.pt[chave];
    if (valor === undefined) return chave;

    if (typeof valor === 'object') {
      const categoria = new Intl.PluralRules(idioma).select(vars.n);
      valor = valor[categoria] ?? valor.other;
    }

    return valor.replace(/\{(\w+)\}/g, (marcador, nome) => (nome in vars ? vars[nome] : marcador));
  }

  function aplicar(idioma) {
    document.documentElement.lang = idioma;
    document.querySelectorAll('[data-i18n]').forEach(el => {
      el.textContent = t(idioma, el.dataset.i18n);
    });
  }

  aplicar('en');
</script>
