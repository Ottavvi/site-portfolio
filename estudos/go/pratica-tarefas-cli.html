Prática: lista de tarefas no terminal

Aula · 30 min

go run . add "Comprar café"
go run . add "Pagar conta de luz"
go run . list         # [ ] 1 Comprar café / [ ] 2 Pagar conta de luz
go run . done 1
go run . list         # [x] 1 Comprar café ...

Tarefas

  1. Struct Tarefa com ID, Titulo, Feita e CriadaEm.
  2. Funções carregar() e salvar() que leem e gravam tarefas.json (arquivo inexistente = lista vazia).
  3. Subcomandos add, list e done, com erros claros.
  4. O ID de uma tarefa nova é o maior ID existente mais 1.
Ver a resposta 
package main

import (
    "encoding/json"
    "errors"
    "fmt"
    "io/fs"
    "os"
    "strconv"
    "time"
)

type Tarefa struct {
    ID       int       `json:"id"`
    Titulo   string    `json:"titulo"`
    Feita    bool      `json:"feita"`
    CriadaEm time.Time `json:"criada_em"`
}

const arquivo = "tarefas.json"

func carregar() ([]Tarefa, error) {
    dados, err := os.ReadFile(arquivo)
    if errors.Is(err, fs.ErrNotExist) {
        return nil, nil
    }
    if err != nil {
        return nil, err
    }
    var tarefas []Tarefa
    return tarefas, json.Unmarshal(dados, &tarefas)
}

func salvar(tarefas []Tarefa) error {
    dados, err := json.MarshalIndent(tarefas, "", "  ")
    if err != nil {
        return err
    }
    return os.WriteFile(arquivo, dados, 0644)
}

func executar(args []string) error {
    if len(args) == 0 {
        return errors.New("uso: tarefas add|list|done")
    }
    tarefas, err := carregar()
    if err != nil {
        return fmt.Errorf("lendo %s: %w", arquivo, err)
    }
    switch args[0] {
    case "add":
        if len(args) < 2 {
            return errors.New("informe o título")
        }
        maior := 0
        for _, t := range tarefas {
            maior = max(maior, t.ID)
        }
        tarefas = append(tarefas, Tarefa{ID: maior + 1, Titulo: args[1], CriadaEm: time.Now()})
        return salvar(tarefas)
    case "list":
        for _, t := range tarefas {
            marca := " "
            if t.Feita {
                marca = "x"
            }
            fmt.Printf("[%s] %d %s\n", marca, t.ID, t.Titulo)
        }
        return nil
    case "done":
        if len(args) < 2 {
            return errors.New("informe o id")
        }
        id, err := strconv.Atoi(args[1])
        if err != nil {
            return fmt.Errorf("id inválido: %s", args[1])
        }
        for i := range tarefas {
            if tarefas[i].ID == id {
                tarefas[i].Feita = true
                return salvar(tarefas)
            }
        }
        return fmt.Errorf("tarefa %d não encontrada", id)
    }
    return fmt.Errorf("comando desconhecido: %s", args[0])
}

func main() {
    if err := executar(os.Args[1:]); err != nil {
        fmt.Fprintln(os.Stderr, "erro:", err)
        os.Exit(1)
    }
}

No done, o laço usa o índice (tarefas[i].Feita = true): com for _, t := range, o t é uma cópia e a mudança se perderia. Separar executar do main deixa a lógica testável, devolvendo erros em vez de sair do programa no meio.
